import { apiService } from "./service.js";

export class EmployeeTable {
    // 1.set your initialStates
    #currentPage = 1;
    #limit = 5;

    #employees = [];
    #pagination = {};

    constructor({
        tableBody,
        paginationContainer
    }) {
        // 2. Get the required data's Via API(internal i.e JS)
        this.tableBody = document.querySelector(tableBody);
        this.paginationContainer = document.querySelector(paginationContainer);
    }

    async init() {
        await this.loadEmployees();
        this.renderTable();
    }

    async loadEmployees() {
        // 2. Get the required data's Via API(external)
        const response = await apiService.get(`/protected/employees-show.php?page=${this.#currentPage}&limit=${this.#limit}`)
        this.#employees = response.data;
        this.#pagination = response.pagination;
        console.log(this.#employees);
        console.log(this.#pagination);
    }

    renderTable() {
        // console.log(this.#employees);
        this.tableBody.innerHTML = "";
        this.#employees.forEach(employee => {
            this.tableBody.insertAdjacentHTML(
                "beforeend",
            `
            <tr>
            <td>${employee.id}</td>
            <td>${employee.employee_id}</td>
            <td>${employee.name}</td>
            <td>${employee.department}</td>
            </tr>
            `)
        })
    }
}

so now know why we are doing like this way i.e API -> loadEmployees() -> storing it as state -> renderTable() without directely doing API -> renderTable() bcos later we will be having deleteEmployee, SearchEmployee, SortEmployee, UpdateEmployee like this and so on so if we keep the data as state like this this.#employee and simply call it using this.renderTable() thats all

bindEvents() {
        this.paginationContainer.addEventListener("click", async (event) => {
            const button = event.target.closest(".page-btn");
            const firstButton = event.target.closest(".pagination-first");
            const prevButton = event.target.closest(".pagination-prev");
            const lastButton = event.target.closest(".pagination-last");
            const nextButton = event.target.closest(".pagination-next");
            if (
                nextButton &&
                this.#currentPage < this.#pagination.totalPages
            ) {
                await this.handlePageChange(
                    this.#currentPage + 1
                );

                return;
            }

            if (
                prevButton &&
                this.#currentPage > 1
            ) {
                await this.handlePageChange(
                    this.#currentPage - 1
                );

                return;
            }

            if(firstButton) {
                await this.handlePageChange(1);
                return;
            }

            if(lastButton) {
                await this.handlePageChange(this.#pagination.totalPages);
                return;
            }
            if (!button) return;
            const page = Number(button.dataset.page);
            await this.handlePageChange(page);
        })
    }
where you can able to see how we handled our 4 buttons only with our own states #currentPage and #pagination.totalPages itself
